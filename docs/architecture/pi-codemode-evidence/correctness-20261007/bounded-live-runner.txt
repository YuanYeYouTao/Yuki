import asyncio,json,logging,sys
from pathlib import Path
import httpx
from scripts.verify_deepseek_codemode import read_credentials
from tools.cache_probe import run_experiment,load_manifest,PhysicalRequests
from qq_ai_bot.model_runtime.models import ModelProfile,ModelProtocol,ModelCapability
from qq_ai_bot.domain.messages import ReasoningEffort
logging.getLogger().setLevel(logging.CRITICAL)
credentials=read_credentials(Path('<private-approved-credential-file>'))
profile=ModelProfile(id='correctness-20261007-synthetic',provider='deepseek',protocol=ModelProtocol.CHAT_COMPLETIONS,base_url=credentials['base_url (openai)'].rstrip('/')+'/',api_key_env='UNUSED',model=credentials['model'],timeout_seconds=120,max_retries=1,default_temperature=0.5,default_max_output_tokens=8192,reasoning_effort=ReasoningEffort.HIGH,capabilities={ModelCapability.TOOLS,ModelCapability.REASONING,ModelCapability.STRUCTURED_OUTPUT})
original_request=PhysicalRequests.request
async def bounded_request(self,request):
    if len(self.rows)>=96 or len(request.content)>400000:
        raise httpx.RequestError('isolated acceptance request bound reached',request=request)
    return await original_request(self,request)
PhysicalRequests.request=bounded_request
async def main():
    output=Path('docs/architecture/pi-codemode-evidence/correctness-20261007/live-cache.json')
    report=await run_experiment(profile=profile,api_key=credentials['api_key'],output=output,scenarios=('C01','C02','C03','C04','C05','C06'),samples=3,warmups=1,prefix_characters=12000,delay_seconds=1,manifest=load_manifest(Path('docs/architecture/pi-codemode-evidence/correctness-20261007/cache-live-manifest.json')),compaction_mode='ordinary')
    report['acceptance_authorization']='User previously explicitly authorized real DeepSeek test calls in this thread; synthetic bounded acceptance only'
    report['request_bounds']={'physical_calls':96,'request_bytes':400000,'output_tokens_each':8192,'max_retries':1}
    report['physical_source']='final provider body and response; upstream final hop unavailable; nonstreaming'
    output.write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
    print(json.dumps({'scenarios':[(s['scenario'],s['status']) for s in report['scenarios']],'physical_calls':len(report['physical_calls']),'totals':report['totals'],'error_category':report.get('experiment_error_category')},ensure_ascii=False))
    if report.get('experiment_error_category') or any(s['status']!='complete' for s in report['scenarios']):
        raise SystemExit(1)
asyncio.run(main())
